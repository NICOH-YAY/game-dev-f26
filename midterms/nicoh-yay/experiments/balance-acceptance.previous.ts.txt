import { beforeAll, it, expect } from "vitest";
import { readFileSync } from "node:fs";
import { AlwaysStrike, SimpleBot, Expectimax, RandomLegal, PersonaController, type Controller } from "../../src/agents/index.js";
import { type Scenario, type Persona } from "../../src/game.js";
import { runEpisode } from "../../src/runner/episode.js";
import { streams } from "../../src/sim/random.js";
const suite = JSON.parse(readFileSync(new URL("../../experiments/suite.json", import.meta.url), "utf8")) as {
 testimony_seed_base: number; situations: (Scenario & { id: string; persona: Persona; duel_seed_base: number })[];
};
const singles = ["always-strike", "always-bash", "always-guard", "always-recover", "always-sidestep"] as const;
const cells = new Map<string, Record<string, number>>();
beforeAll(async () => {
 for (const situation of suite.situations) {
  const personas = new Set<Persona>([situation.persona, "Adapter"]);
  for (const persona of personas) {
   const wins: Record<string, number> = {};
   for (const name of ["baseline", "random", ...singles]) {
    wins[name] = 0;
    for (let i = 0; i < 20; i++) {
     const seed = situation.duel_seed_base + i;
     const controller: Controller = name === "baseline" ? new Expectimax() : name === "random" ? new RandomLegal(seed ^ 0x12345678) : name === "always-strike" ? new AlwaysStrike() : new SimpleBot(name as ConstructorParameters<typeof SimpleBot>[0]);
     const result = await runEpisode({seed, scenario: {...situation, persona, testimonySeed: suite.testimony_seed_base + i}, controller, opponent: new PersonaController(persona, streams(seed).persona)});
     wins[name] += Number(result.outcome.winner === 0);
    }
   }
   cells.set(`${situation.id}/${persona}`, wins);
  }
 }
}, 60000);
for (const situation of suite.situations) {
 it(`criterion (a): ${situation.id}/${situation.persona} baseline >= every single-action bot and > random`, () => {
  const wins = cells.get(`${situation.id}/${situation.persona}`)!;
  expect.soft(wins.baseline, "baseline > random").toBeGreaterThan(wins.random);
  for (const name of singles) expect.soft(wins.baseline, `baseline >= ${name} (${wins[name]})`).toBeGreaterThanOrEqual(wins[name]);
 });
 it(`criterion (b): ${situation.id}/Adapter every single-action bot <= 10/20`, () => {
  const wins = cells.get(`${situation.id}/Adapter`)!;
  for (const name of singles) expect.soft(wins[name], name).toBeLessThanOrEqual(10);
 });
}
